% 5. SAE explainer
\begin{frame}{Sparse Autoencoder (SAEs)}
  \vspace{0.04cm}
  {\normalsize\color{inktwo}
  An SAE learns a wide dictionary of directions over a model's activations. A sparse encoder maps each activation to only a handful of active entries; the decoder reconstructs the original. Sparsity forces each dictionary entry to be monosemantic.\par}

  \vspace{0.4cm}
  \begin{center}
    \includegraphics[width=0.84\textwidth,height=0.42\textheight,keepaspectratio]{SAE_diagram.png}
  \end{center}

  \vspace{0.22cm}
  {\centering\color{accent}\normalsize\textbf{This gives us a feature-level vocabulary for the residual stream}\par}
  \vspace{0.1cm}
  {\centering\color{muted}\scriptsize Building on Bricken et al. (2023) and Templeton et al. (2024)\par}
\end{frame}

% 13. Limitations
\begin{frame}{Limitations}
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.32\textwidth}
    \begin{beamercolorbox}[wd=\linewidth,sep=0.24cm,rounded=true]{normal text}
      \textbf{Correlational evidence}\\[0.08cm]
      \small The analysis links feature shifts to behavior, but it does not yet prove that those features causally drive the behavior.
    \end{beamercolorbox}

    \column{0.32\textwidth}
    \begin{beamercolorbox}[wd=\linewidth,sep=0.24cm,rounded=true]{normal text}
      \textbf{Narrow experimental scope}\\[0.08cm]
      \small Results come from one matched PT/IT model pair, one architecture scale, and one layer rather than a broad cross-model study.
    \end{beamercolorbox}

    \column{0.32\textwidth}
    \begin{beamercolorbox}[wd=\linewidth,sep=0.24cm,rounded=true]{normal text}
      \textbf{Imperfect semantic labeling}\\[0.08cm]
      \small Feature taxonomy is useful but noisy: labels are partly interpretive, and inter-annotator agreement reached about 60\%.
    \end{beamercolorbox}
  \end{columns}

% 14. Conclusion
\begin{frame}{Conclusion}
  \begin{beamercolorbox}[wd=\linewidth,sep=0.3cm,rounded=true]{normal text}
    {\Large\color{ink}\textbf{Main takeaway}}\\[0.14cm]
    {\normalsize\color{inktwo}Post-training appears to act less by creating entirely new internal structure and more by \textbf{reorganizing, amplifying, and sharpening existing pre-training features}.}
  \end{beamercolorbox}

  \vspace{0.35cm}
  \begin{itemize}
    \item Modified features dominate the matched dictionaries, while preserved features remain a small minority.
    \item The strongest instruction-following and safety-related IT features mostly trace back to shared PT substrate rather than to purely IT-exclusive features.
    \item The evidence therefore supports a \textbf{weak but consistent claim}: behaviorally central post-training features are usually transformed versions of pre-training features.
  \end{itemize}
\end{frame}
